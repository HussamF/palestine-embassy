import { useState, useEffect } from 'react';
import { FaPlay, FaImages } from 'react-icons/fa';

  id: number;
  tag: string;
  title: string;
  excerpt: string;
  date: string;
  hasVideo?: boolean;
  image?: string;    // للصورة المفردة
  images?: string[];  // لمجموعة الصور
}

// --- 2. قاعدة البيانات التجريبية ---
const newsData: NewsItem[] = [
  {
    id: 1,
    tag: "لقاءات وبيانات السفير",
    title: "سفارة دولة فلسطين تطلق منتدى الاستثمار الاقتصادي المشترك في الدوحة",
    excerpt: "بحضور وفود رفيعة المستوى، ناقش المنتدى فرص التعاون الثنائي لتعزيز الاستثمار ودعم المبادرات التنموية لعام 2026.",
    date: "18 يناير",
    hasVideo: true,
    images: [
      "/imgs/news/main-news.jpg",

    ]
  },
  {

    date: "16 يناير",

  },
  {
    id: 4,
    tag: "لقاءات وبيانات السفير",

    excerpt: "نحيطكم علماً بمواعيد إغلاق وفتح القسم القنصلي خلال فترة الأعياد الرسمية.",
    date: "14 يناير",
    image: "/imgs/news/sub-news1.jpg"
  }
];

export default function News() {
  const [currentSlide, setCurrentSlide] = useState(0);

  // --- 3. منطق التوزيع ---
  const mainNews = newsData.find(n => n.id === 1) || newsData[0];
  const otherNews = newsData.filter(n => n.id !== 1).slice(0, 5);

  // مصفوفة الصور للخبر المميز
  const sliderImages = mainNews.images || (mainNews.image ? [mainNews.image] : []);

  // --- 4. مؤقت السلايدر ---
  useEffect(() => {
    if (sliderImages.length > 1) {
      const timer = setInterval(() => {
        setCurrentSlide((prev) => (prev + 1) % sliderImages.length);
      }, 4000);
      return () => clearInterval(timer);
    }
  }, [sliderImages.length]);

  return (
    <section id="news" className="py-20 bg-slate-50" dir="rtl">
      <div className="max-w-7xl mx-auto px-6">
        
        <h2 className="text-4xl font-black text-[#1e1b4b] mb-12 text-right border-r-8 border-[#c5a059] pr-5">آخر الأخبار والفعاليات</h2>

        <div className="flex flex-col lg:flex-row gap-10">
          
          {/* القسم الأيمن: الخبر المميز */}
          <div className="w-full lg:w-2/3">
            <article className="group bg-white rounded-[3rem] overflow-hidden shadow-2xl border border-white flex flex-col h-full transition-all">
              
              <div className="relative h-[350px] md:h-[500px] overflow-hidden bg-slate-200">
                {sliderImages.map((img, idx) => (
                  <img
                    key={idx}
                    src={img}
                    className={`absolute inset-0 w-full h-full object-cover transition-opacity duration-1000 group-hover:scale-105 ${
                      idx === currentSlide ? 'opacity-100' : 'opacity-0'
                    }`}
                    alt="News Image"
                  />
                ))}
                
                {/* شارات الوسائط */}
                <div className="absolute top-6 left-6 flex gap-2 z-30">
                   {mainNews.hasVideo && (
                     <div className="bg-red-600 text-white p-3 rounded-full shadow-lg animate-pulse">
                        <FaPlay size={12} />
                     </div>
                   )}
                   {sliderImages.length > 1 && (
                     <div className="bg-[#1e1b4b] text-white p-3 rounded-full shadow-lg">
                        <FaImages size={12} />
                     </div>
                   )}
                </div>

                <div className="absolute top-6 right-6 px-6 py-2 rounded-full text-xs font-black z-30 bg-white/95 text-[#c5a059] shadow-lg">
                  {mainNews.tag}
                </div>

                {sliderImages.length > 1 && (
                  <div className="absolute bottom-6 left-1/2 -translate-x-1/2 flex gap-2 z-30 bg-black/20 backdrop-blur-md p-2 rounded-full">
                    {sliderImages.map((_, idx) => (
                      <div 
                        key={idx} 
                        className={`h-1.5 rounded-full transition-all duration-500 ${idx === currentSlide ? 'w-8 bg-[#c5a059]' : 'w-2 bg-white/50'}`} 
                      />
                    ))}
                  </div>
                )}
              </div>

              <div className="p-10 text-right">
                <span className="text-slate-400 font-bold mb-2 block italic">{mainNews.date} 2026</span>
                <h3 className="text-3xl font-black text-[#1e1b4b] mb-4 group-hover:text-[#c5a059] transition-colors">{mainNews.title}</h3>
                <p className="text-slate-600 text-lg line-clamp-2 mb-6 leading-relaxed font-medium">{mainNews.excerpt}</p>
                <button className="bg-slate-50 text-[#1e1b4b] font-black px-8 py-4 rounded-2xl hover:bg-[#c5a059] hover:text-white transition-all flex items-center gap-3 w-fit">
                  اقرأ التفاصيل <span>←</span>
                </button>
              </div>
            </article>
          </div>

          {/* القسم الأيسر: القائمة الجانبية */}
          <div className="w-full lg:w-1/3 flex flex-col gap-5">
            <h4 className="text-xl font-black text-[#1e1b4b] mb-2 flex items-center gap-2">
               <span className="w-2 h-6 bg-[#c5a059] rounded-full"></span>
               أخبار سابقة
            </h4>
            
            {otherNews.map((item) => (
              <div key={item.id} className="bg-white p-4 rounded-[2rem] border border-slate-100 shadow-sm flex gap-4 items-center group transition-all hover:shadow-xl cursor-pointer">
                <div className="w-24 h-24 rounded-2xl overflow-hidden shrink-0 relative bg-slate-100">
                  <img 
                    // الحل النهائي لمشكلة الصور: التحقق من image أولاً ثم من مصفوفة images
                    src={item.image || (item.images && item.images[0]) || "/imgs/placeholder.jpg"} 
                    className="w-full h-full object-cover group-hover:scale-110 transition-transform duration-500" 
                    alt="thumbnail" 
                  />
                  {item.hasVideo && (
                    <div className="absolute inset-0 bg-black/30 flex items-center justify-center text-white">
                      <FaPlay size={10} />
                    </div>
                  )}
                </div>
                <div className="text-right flex-1">
                  <span className={`text-[9px] font-black px-2 py-1 rounded-lg mb-2 inline-block 
                    ${item.tag === "إخطارات" ? "bg-red-50 text-red-600" : "bg-amber-50 text-[#c5a059]"}`}>
                    {item.tag}
                  </span>
                  <h5 className="font-bold text-[#1e1b4b] text-sm line-clamp-2 leading-snug group-hover:text-[#c5a059] transition-colors">
                    {item.title}
                  </h5>
                </div>
              </div>
            ))}

            <button className="w-full py-5 border-2 border-dashed border-slate-200 rounded-[2rem] text-slate-400 font-black hover:border-[#c5a059] hover:text-[#c5a059] transition-all mt-4 text-sm">
               مشاهدة جميع الأخبار
            </button>
          </div>

        </div>
      </div>
    </section>
  );
}
